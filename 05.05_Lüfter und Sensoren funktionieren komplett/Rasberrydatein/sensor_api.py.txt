from flask import Flask, jsonify
from flask_cors import CORS
import board
import adafruit_dht
import RPi.GPIO as GPIO
import time

app = Flask(__name__)
CORS(app)

# DHT11 an GPIO4 / Pin 7
dht = adafruit_dht.DHT11(board.D4)

# Motor / Lï¿½fter an GPIO17 / Pin 11
MOTOR_PIN = 17
GRENZWERT_TEMP = 18

GPIO.setmode(GPIO.BCM)
GPIO.setup(MOTOR_PIN, GPIO.OUT)
GPIO.output(MOTOR_PIN, GPIO.LOW)

def lese_sensor():
    for i in range(3):
        try:
            temperatur = dht.temperature
            luft = dht.humidity

            if temperatur is not None and luft is not None:
                return temperatur, luft

        except RuntimeError:
            time.sleep(2)

    return None, None

@app.route("/data")
def data():
    temperatur, luft = lese_sensor()

    if temperatur is not None:
        if temperatur > GRENZWERT_TEMP:
            GPIO.output(MOTOR_PIN, GPIO.HIGH)
            motor_status = "AN"
            motor_grund = "Temperatur ï¿½ber Grenzwert"
        else:
            GPIO.output(MOTOR_PIN, GPIO.LOW)
            motor_status = "AUS"
            motor_grund = "Temperatur unter Grenzwert"
    else:
        GPIO.output(MOTOR_PIN, GPIO.LOW)
        motor_status = "AUS"
        motor_grund = "Keine gï¿½ltigen Sensordaten"

    return jsonify({
        "temperatur": temperatur,
        "luftfeuchtigkeit": luft,
        "motor": motor_status,
        "motor_grund": motor_grund,
        "grenzwert": GRENZWERT_TEMP
    })

@app.route("/")
def startseite():
    return jsonify({
        "status": "Sensor API lï¿½uft",
        "endpunkt": "/data"
    })

try:
    app.run(host="0.0.0.0", port=5000)
finally:
    GPIO.output(MOTOR_PIN, GPIO.LOW)
    GPIO.cleanup()
